# --description--

Add two more columns after `event`. `actor_id` is a whole number that may be empty. When it is there, it must match a real user: the person who did it. `at` is a `timestamptz` that must have a value and is `now()` unless you say otherwise. With this step your database has all seven tables.

A new history entry should get the time it was made, and it should be saved with no `actor_id`. Add `default now()` to `at`.

```js

  await db.query(
    "insert into hostels (name, gender, level) values ('H22a', 'female', 100) returning id",

).rows[0].id;

  await db.query('insert into rooms (hostel_id, number) values ($1, $2) returning id', [h, 'R22a'])

  await db.query(
    "insert into users (full_name, email, password_hash, role) values ('User 22a', 'user22a@example.com', 'x', 'student') returning id",
  )

const c = (
  await db.query(
    'insert into complaints (student_id, room_id, description, category, urgency, deadline) values ($1, $2, $3, $4, $5, now()) returning id',

const { rows } = await db.query('insert into complaint_history (complaint_id, event) values ($1, $2) returning actor_id, at', [c, 'Test']);

assert.exists(rows[0].at);
```

```sql file=query.sql
-- Hostel blocks.
create table hostels (
  id integer generated always as identity primary key,
  name text not null unique,
  gender text not null check (gender in ('male', 'female')),
  level integer not null
);

-- Rooms. Each room belongs to one hostel.
create table rooms (
  id integer generated always as identity primary key,
  hostel_id integer not null references hostels(id),
  number text not null,
  unique (hostel_id, number)
);

-- Bed spaces. Each bed space is one bed in a room.
create table bed_spaces (
  id integer generated always as identity primary key,
  room_id integer not null references rooms(id),
  label text not null,
  unique (room_id, label)
);

-- Users: students, maintenance officers, hall wardens and admins.
create table users (
  id integer generated always as identity primary key,
  full_name text not null,
  email text not null unique,
  password_hash text not null,
  role text not null check (role in ('student', 'maintenance_officer', 'hall_warden', 'admin')),
  gender text check (gender in ('male', 'female')),
  level integer,
  fees_cleared boolean not null default false,
  unit text,                                -- maintenance officers: plumbing, electrical, ...
  hostel_id integer references hostels(id)  -- hall wardens: the hostel they run
);

-- Allocations: a student given a bed space.
create table allocations (
  id integer generated always as identity primary key,
  student_id integer not null references users(id),
  bed_space_id integer not null references bed_spaces(id),
  status text not null default 'active' check (status in ('active', 'ended')),
  created_at timestamptz not null default now()
);

-- A bed can have only one active allocation.
create unique index one_active_allocation_per_bed
  on allocations (bed_space_id) where status = 'active';

-- A student can have only one active allocation.
create unique index one_active_allocation_per_student
  on allocations (student_id) where status = 'active';

-- Complaints about a room.
create table complaints (
  id integer generated always as identity primary key,
  student_id integer not null references users(id),
  room_id integer not null references rooms(id),
  description text not null,
  category text not null,
  urgency text not null check (urgency in ('urgent', 'normal', 'low')),
  status text not null default 'Open' check (status in ('Open', 'Assigned', 'In Progress', 'Resolved')),
  escalation_level integer not null default 0,
  created_at timestamptz not null default now(),
  deadline timestamptz not null,
  resolved_at timestamptz
);

-- The history of each complaint.
--editable-region--
create table complaint_history (
  id integer generated always as identity primary key,
  complaint_id integer not null references complaints(id),
  event text not null
);
--editable-region--
```
