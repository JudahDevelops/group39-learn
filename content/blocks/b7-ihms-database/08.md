# --description--

A room cannot have two bed spaces with the same label. Add `unique (room_id, label)` after the columns.

# --hints--

Two bed spaces in the same room should not share a label. Add `unique (room_id, label)` after the columns.

```js
const h = (
  await db.query(
    "insert into hostels (name, gender, level) values ('S8a', 'female', 100) returning id",
  )
).rows[0].id;
const r = (
  await db.query('insert into rooms (hostel_id, number) values ($1, $2) returning id', [h, 'S8-1'])
).rows[0].id;
await db.query('insert into bed_spaces (room_id, label) values ($1, $2)', [r, '1']);
let error;
try {
  await db.query('insert into bed_spaces (room_id, label) values ($1, $2)', [r, '1']);
} catch (e) {
  error = e;
}
assert.exists(error);
assert.equal(error.code, '23505');
```

Two different rooms should be able to have a bed space with the same label.

```js
const h = (
  await db.query(
    "insert into hostels (name, gender, level) values ('S8b', 'female', 100) returning id",
  )
).rows[0].id;
const r1 = (
  await db.query('insert into rooms (hostel_id, number) values ($1, $2) returning id', [h, 'S8-2'])
).rows[0].id;
const r2 = (
  await db.query('insert into rooms (hostel_id, number) values ($1, $2) returning id', [h, 'S8-3'])
).rows[0].id;
await db.query('insert into bed_spaces (room_id, label) values ($1, $2)', [r1, '1']);
await db.query('insert into bed_spaces (room_id, label) values ($1, $2)', [r2, '1']);
```

  name text not null unique,
  gender text not null check (gender in ('male', 'female')),
  level integer not null
);

-- Rooms. Each room belongs to one hostel.

  id integer generated always as identity primary key,
  hostel_id integer not null references hostels(id),

);
