-- Books in the hostel library.
create table books (
  id integer generated always as identity primary key,
  title text not null,
  author text not null,
  subject text not null
);

-- Members who borrow books.
create table members (
  id integer generated always as identity primary key,
  full_name text not null,
  room text not null
);

-- Loans: which member has which book.
create table loans (
  id integer generated always as identity primary key,
  book_id integer not null references books(id),
  member_id integer not null references members(id),
  status text not null default 'out' check (status in ('out', 'returned'))
);

-- Add the first book.
insert into books (title, author, subject)
values ('Things Fall Apart', 'Chinua Achebe', 'Fiction');

-- Add the other six books.
insert into books (title, author, subject) values
  ('Half of a Yellow Sun', 'Chimamanda Ngozi Adichie', 'Fiction'),
  ('Purple Hibiscus', 'Chimamanda Ngozi Adichie', 'Fiction'),
  ('Python Crash Course', 'Eric Matthes', 'Computing'),
  ('Clean Code', 'Robert C. Martin', 'Computing'),
  ('Learning SQL', 'Alan Beaulieu', 'Computing'),
  ('Eloquent JavaScript', 'Marijn Haverbeke', 'Computing');

-- Add the members.
insert into members (full_name, room) values
  ('Amina Bello', 'B-214'),
  ('Chidi Okafor', 'A-101'),
  ('Bola Adeyemi', 'A-102'),
  ('Emeka Eze', 'B-101'),
  ('Ngozi Obi', 'B-102');

-- Record the loans.
insert into loans (book_id, member_id, status) values
  (1, 1, 'returned'),
  (2, 1, 'out'),
  (4, 2, 'out'),
  (4, 3, 'returned'),
  (5, 3, 'out'),
  (3, 2, 'returned'),
  (1, 4, 'out');

-- Show every book.
select * from books;

-- Show only the titles and authors.
select title, author from books;

-- Show the titles of the Computing books.
select title from books where subject = 'Computing';

-- Show every title, A to Z.
select title from books order by title;

-- Show the first three titles.
select title from books order by title limit 3;

-- How many loans are still out?
select count(*) as loans_out from loans where status = 'out';

-- Amina returns Half of a Yellow Sun, which is loan 2.
update loans set status = 'returned' where id = 2;

-- Show each loan with the title of its book.
select l.id, b.title, l.status
from loans l
join books b on b.id = l.book_id
order by l.id;

-- Show who has each book that is out.
select m.full_name as member, b.title as book, m.room
from loans l
join books b on b.id = l.book_id
join members m on m.id = l.member_id
where l.status = 'out'
order by m.full_name;

-- Count the books in each subject.
select subject, count(*) as books
from books
group by subject
order by subject;

-- Members with two or more loans.
select m.full_name, count(l.id) as loans
from members m
left join loans l on l.member_id = m.id
group by m.full_name
having count(l.id) >= 2
order by m.full_name;

## --hint--

Start from `loans l`, join `books b`, and group by the book's subject.
