# --description--

Right now nothing stops the librarian from lending a book that is already out. Make the database refuse it. Create a unique index called `one_loan_out_per_book` on `loans (book_id)` that covers only the loans with the status `out`. Returned loans must not count.

-- Books in the hostel library.
create table books (
  id integer generated always as identity primary key,
  title text not null,
  author text not null,
  subject text not null
);

-- Members who borrow books.
create table members (
  id integer generated always as identity primary key,
  full_name text not null,
  room text not null
);

-- Loans: which member has which book.
create table loans (
  id integer generated always as identity primary key,
  book_id integer not null references books(id),
  member_id integer not null references members(id),
  status text not null default 'out' check (status in ('out', 'returned'))
);

-- Add the first book.
insert into books (title, author, subject)
values ('Things Fall Apart', 'Chinua Achebe', 'Fiction');

-- Add the other six books.

  ('Clean Code', 'Robert C. Martin', 'Computing'),
  ('Learning SQL', 'Alan Beaulieu', 'Computing'),
  ('Eloquent JavaScript', 'Marijn Haverbeke', 'Computing');

  ('Amina Bello', 'B-214'),
  ('Chidi Okafor', 'A-101'),
  ('Bola Adeyemi', 'A-102'),
  ('Emeka Eze', 'B-101'),
  ('Ngozi Obi', 'B-102');

-- Record the loans.
insert into loans (book_id, member_id, status) values
  (1, 1, 'returned'),
  (2, 1, 'out'),
  (4, 2, 'out'),
  (4, 3, 'returned'),
  (5, 3, 'out'),
  (3, 2, 'returned'),
  (1, 4, 'out');

-- Show every book.
select * from books;

select title from books order by title;
