# --description--

A new complaint needs an id. The server will give them out later. Until then the page numbers them itself.

Below `render`, create `let complaints = [];`. It will hold every complaint on the page. Then write `nextId()`. It loops over `complaints` and keeps the highest `c.id` in a `let` called `highest`, which starts at 0. It returns `highest + 1`.

```js
assert.isArray(complaints);
```

Your code should define a function called `nextId`.

```js
assert.isFunction(nextId);
```

With no complaints, `nextId()` should return `1`.


```js file=js/student.js
// Triage rules. The urgency keywords come from Table IV of the paper.

const URGENCY_RULES = [
  { urgency: "urgent", keywords: ["burst", "flood", "spark", "exposed wire", "no water", "broken lock"] },
  { urgency: "normal", keywords: ["leaking tap", "faulty socket", "broken bed", "blocked drain"] },
  { urgency: "low", keywords: ["paint", "loose handle", "squeaking door"] },
];

const CATEGORY_RULES = [
  { category: "plumbing", keywords: ["pipe", "tap", "leak", "water", "drain", "toilet", "shower", "flood"] },
  { category: "electrical", keywords: ["socket", "light", "bulb", "wire", "spark", "fan", "switch", "power"] },
  { category: "carpentry", keywords: ["door", "bed", "window", "wardrobe", "handle", "lock", "chair", "table"] },
  { category: "sanitation", keywords: ["dirty", "smell", "refuse", "waste", "cockroach", "rat", "bin"] },
];

// Helpers from the lesson on the DOM and events.
function cell(text) {
  const td = document.createElement("td");
  td.textContent = text;
  return td;
}

function badge(urgency) {
  const td = document.createElement("td");
  const span = document.createElement("span");
  span.classList.add("badge", urgency);
  span.textContent = urgency;
  td.append(span);
  return td;
}

const form = document.querySelector("#complaint-form");
const descriptionBox = document.querySelector("#description");
const message = document.querySelector("#message");

function triageUrgency(description) {
  const text = description.toLowerCase();
  for (const rule of URGENCY_RULES) {
    for (const keyword of rule.keywords) {
      if (text.includes(keyword)) {
        return rule.urgency;
      }
    }
  }
  return "normal";
}

function triageCategory(description) {
  const text = description.toLowerCase();
  for (const rule of CATEGORY_RULES) {
    for (const keyword of rule.keywords) {
      if (text.includes(keyword)) {
        return rule.category;
      }
    }
  }
  return "other";
}

function deadlineHours(urgency) {
  if (urgency === "urgent") return 24;
  if (urgency === "normal") return 72;
  return 168;
}

const HOUR = 60 * 60 * 1000;

function hoursUntil(deadline) {
  return (new Date(deadline) - new Date()) / HOUR;
}

function dueLabel(deadline) {
  const hoursLeft = hoursUntil(deadline);
  return hoursLeft >= 0
    ? `Due in ${Math.round(hoursLeft)} h`
    : `Overdue by ${Math.round(-hoursLeft)} h`;
}

const tbody = document.querySelector("#complaints tbody");

function render(list) {
  tbody.replaceChildren();
  for (const c of list) {
    const row = document.createElement("tr");
    row.append(
      cell(c.id),
      cell(c.category),
      badge(c.urgency),
      cell(new Date(c.deadline).toLocaleString("en-NG")),
      cell(dueLabel(c.deadline)),
      cell(c.status)
    );
    if (hoursUntil(c.deadline) < 0) {
      row.classList.add("overdue");
    }
    tbody.append(row);
  }
}
```
