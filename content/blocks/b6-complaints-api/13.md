# --description--

Add two keys to `complaint`: `category` and `urgency`, taken from `result`. When no keyword matches, your `triage` uses the placeholder urgency `normal`. What should happen then is one of the six open design decisions. See <a href="/reference/decisions" target="_blank" rel="noopener">Group decisions</a>.

# --hints--

"The pipe in the bathroom has burst" should come back with the category `plumbing`. Use `result["category"]`.

```py
r = await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
assert r.json()['category'] == 'plumbing'
```

"The pipe in the bathroom has burst" should come back with the urgency `urgent`. Use `result["urgency"]`.

```py
r = await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
assert r.json()['urgency'] == 'urgent'
```

"There's a spark coming from the socket" should be `electrical` and `urgent`.

```py
r = await client.post('/api/complaints', json={'description': "There's a spark coming from the socket"})
assert r.json()['category'] == 'electrical'
assert r.json()['urgency'] == 'urgent'
```

"The squeaking door keeps me awake" should be `carpentry` and `low`.

```py
r = await client.post('/api/complaints', json={'description': 'The squeaking door keeps me awake'})
assert r.json()['category'] == 'carpentry'
assert r.json()['urgency'] == 'low'
```

A complaint with no keyword, such as "My room is too hot at night", should still be lodged with the category `other` and a valid urgency.

```py
r = await client.post('/api/complaints', json={'description': 'My room is too hot at night'})
assert r.status_code == 201
assert r.json()['category'] == 'other'
assert r.json()['urgency'] in ('urgent', 'normal', 'low')
```

# IHMS triage rules, from chapter B4


def deadline_hours(urgency: str) -> int:
    """Return the resolution deadline in hours for an urgency level."""
    if urgency == "urgent":
        return 24
    if urgency == "normal":
        return 72
    return 168


URGENCY_KEYWORDS = {
    "urgent": ["burst", "flood", "spark", "exposed wire", "no water", "broken lock"],
    "normal": ["leaking tap", "faulty socket", "broken bed", "blocked drain"],
    "low": ["paint", "loose handle", "squeaking door"],
}


def triage_urgency(description: str) -> str | None:
    text = description.lower()
    for urgency in ["urgent", "normal", "low"]:      # most serious first
        for keyword in URGENCY_KEYWORDS[urgency]:
            if keyword in text:
                return urgency
    return None   # nothing matched: what happens next is a group decision


CATEGORY_KEYWORDS = {
    "plumbing": ["pipe", "tap", "leak", "water", "drain", "toilet", "shower", "flood"],
    "electrical": ["socket", "light", "bulb", "wire", "spark", "fan", "switch", "power"],
    "carpentry": ["door", "bed", "window", "wardrobe", "handle", "lock", "chair", "table"],
    "sanitation": ["dirty", "smell", "refuse", "waste", "cockroach", "rat", "bin"],
}


def triage_category(description: str) -> str:
    text = description.lower()
    for category in ["plumbing", "electrical", "carpentry", "sanitation"]:
        for keyword in CATEGORY_KEYWORDS[category]:
            if keyword in text:
                return category
    return "other"


# Placeholder until the group decides: the urgency when no keyword matches
NO_KEYWORD_URGENCY = "normal"


def triage(description: str) -> dict:
    category = triage_category(description)
    urgency = triage_urgency(description)
    if urgency is None:
        urgency = NO_KEYWORD_URGENCY
    return {
        "category": category,
        "urgency": urgency,
        "deadline_hours": deadline_hours(urgency),
    }
