# --description--

Record the change. In `advance_complaint`, after the status changes: get the current time in `now`; when the new status is `Resolved`, set the complaint's `resolved_at` to `now`; and append a history entry whose `event` is `Status changed to` plus the new status, with `actor_id` `None` and `at` `now`. That is the last endpoint. Later chapters move the complaints into PostgreSQL and add login.

# --hints--

`resolved_at` should stay `None` until the complaint is `Resolved`.

```py
new = (await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})).json()
```

When a complaint becomes `Resolved`, `resolved_at` should be set to the current time.

```py
from datetime import datetime, timedelta, timezone
new = (await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})).json()
url = '/api/complaints/' + str(new['id']) + '/advance'
for _ in range(2):
    await client.patch(url)
r = await client.patch(url)
assert r.json()['status'] == 'Resolved'
assert r.json()['resolved_at'] is not None
resolved = datetime.fromisoformat(r.json()['resolved_at'])
assert abs(datetime.now(timezone.utc) - resolved) < timedelta(minutes=1)
```

Each advance should add a history entry `Status changed to <status>`, after the entry made when the complaint was lodged.

```py
new = (await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})).json()
url = '/api/complaints/' + str(new['id']) + '/advance'
await client.patch(url)
r = await client.get('/api/complaints/' + str(new['id']))
events = [h['event'] for h in r.json()['history']]
assert events == ['Lodged and triaged as urgent plumbing', 'Status changed to Assigned']
```

After all three advances, the history should list the four events, oldest first.

```py
new = (await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})).json()
url = '/api/complaints/' + str(new['id']) + '/advance'
for _ in range(3):
    await client.patch(url)
r = await client.get('/api/complaints/' + str(new['id']))
events = [h['event'] for h in r.json()['history']]
assert events == ['Lodged and triaged as urgent plumbing', 'Status changed to Assigned', 'Status changed to In Progress', 'Status changed to Resolved']
```

Each history entry should have `actor_id` set to `None` and an `at` time.

```py
new = (await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})).json()
await client.patch('/api/complaints/' + str(new['id']) + '/advance')
r = await client.get('/api/complaints/' + str(new['id']))
for entry in r.json()['history']:
    assert entry['actor_id'] is None
    assert entry['at']
```

```python file=services/triage.py
# IHMS triage rules, from chapter B4


def deadline_hours(urgency: str) -> int:
    """Return the resolution deadline in hours for an urgency level."""
    if urgency == "urgent":
        return 24
    if urgency == "normal":
        return 72
    return 168


URGENCY_KEYWORDS = {
    "urgent": ["burst", "flood", "spark", "exposed wire", "no water", "broken lock"],
    "normal": ["leaking tap", "faulty socket", "broken bed", "blocked drain"],
    "low": ["paint", "loose handle", "squeaking door"],
}


def triage_urgency(description: str) -> str | None:
    text = description.lower()
    for urgency in ["urgent", "normal", "low"]:      # most serious first
        for keyword in URGENCY_KEYWORDS[urgency]:
            if keyword in text:
                return urgency
    return None   # nothing matched: what happens next is a group decision


CATEGORY_KEYWORDS = {
    "plumbing": ["pipe", "tap", "leak", "water", "drain", "toilet", "shower", "flood"],
    "electrical": ["socket", "light", "bulb", "wire", "spark", "fan", "switch", "power"],
    "carpentry": ["door", "bed", "window", "wardrobe", "handle", "lock", "chair", "table"],
    "sanitation": ["dirty", "smell", "refuse", "waste", "cockroach", "rat", "bin"],
}


def triage_category(description: str) -> str:
    text = description.lower()
    for category in ["plumbing", "electrical", "carpentry", "sanitation"]:
        for keyword in CATEGORY_KEYWORDS[category]:
            if keyword in text:
                return category
    return "other"


# Placeholder until the group decides: the urgency when no keyword matches
NO_KEYWORD_URGENCY = "normal"


def triage(description: str) -> dict:
    category = triage_category(description)
    urgency = triage_urgency(description)
    if urgency is None:
        urgency = NO_KEYWORD_URGENCY
    return {
        "category": category,
        "urgency": urgency,
        "deadline_hours": deadline_hours(urgency),
    }
```
