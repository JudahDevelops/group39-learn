# --description--

The complaint that is due first should come first. Above the list route, write a function `by_deadline(complaint)` that returns `complaint["deadline"]`. Then, in `list_complaints`, return `sorted(complaints, key=by_deadline)` instead of `complaints`. The `key=` tells `sorted` what to compare.

# --hints--

`routers/complaints.py` should define a function `by_deadline(complaint)` that returns the complaint's `deadline`.

```py
import routers.complaints as rc
assert rc.by_deadline({'deadline': 5}) == 5
```

`GET /api/complaints` should list the complaint that is due first, first. A low complaint lodged before an urgent one should come after it.

```py
from datetime import datetime
await client.post('/api/complaints', json={'description': 'Please paint the wall in my room'})
await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
rows = (await client.get('/api/complaints')).json()
deadlines = [datetime.fromisoformat(c['deadline']) for c in rows]
assert deadlines == sorted(deadlines)
```

The filtered list should be sorted by deadline too.

```py
from datetime import datetime

rows = (await client.get('/api/complaints', params={'status': 'Open'})).json()

# IHMS triage rules, from chapter B4


def deadline_hours(urgency: str) -> int:
    """Return the resolution deadline in hours for an urgency level."""
    if urgency == "urgent":
        return 24
    if urgency == "normal":
        return 72
    return 168


URGENCY_KEYWORDS = {
    "urgent": ["burst", "flood", "spark", "exposed wire", "no water", "broken lock"],
    "normal": ["leaking tap", "faulty socket", "broken bed", "blocked drain"],
    "low": ["paint", "loose handle", "squeaking door"],
}


def triage_urgency(description: str) -> str | None:
    text = description.lower()
    for urgency in ["urgent", "normal", "low"]:      # most serious first
        for keyword in URGENCY_KEYWORDS[urgency]:
            if keyword in text:
                return urgency
    return None   # nothing matched: what happens next is a group decision


CATEGORY_KEYWORDS = {
    "plumbing": ["pipe", "tap", "leak", "water", "drain", "toilet", "shower", "flood"],
    "electrical": ["socket", "light", "bulb", "wire", "spark", "fan", "switch", "power"],
    "carpentry": ["door", "bed", "window", "wardrobe", "handle", "lock", "chair", "table"],
    "sanitation": ["dirty", "smell", "refuse", "waste", "cockroach", "rat", "bin"],
}


def triage_category(description: str) -> str:
    text = description.lower()
    for category in ["plumbing", "electrical", "carpentry", "sanitation"]:
        for keyword in CATEGORY_KEYWORDS[category]:
            if keyword in text:
                return category
    return "other"


# Placeholder until the group decides: the urgency when no keyword matches
NO_KEYWORD_URGENCY = "normal"


def triage(description: str) -> dict:
    category = triage_category(description)
    urgency = triage_urgency(description)
    if urgency is None:
        urgency = NO_KEYWORD_URGENCY
    return {
        "category": category,
        "urgency": urgency,
        "deadline_hours": deadline_hours(urgency),
    }
```
