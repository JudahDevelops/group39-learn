# --description--

Save the complaint. Just before `return complaint`, append it to `COMPLAINTS`. Now `GET /api/complaints` lists what you lodged, and each new complaint gets the next id.

# --hints--

A lodged complaint should be saved: `GET /api/complaints` should list one more complaint. Use `COMPLAINTS.append(complaint)`.

```py
before = len((await client.get('/api/complaints')).json())
await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
after = len((await client.get('/api/complaints')).json())
assert after == before + 1
```

Each new complaint should get the next id, one more than the one before.

```py
a = (await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})).json()['id']
b = (await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})).json()['id']
assert b == a + 1
```

The saved complaint should be in the list with the same `id` and `description`.

```py
r = await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
new = r.json()
rows = (await client.get('/api/complaints')).json()
assert any(c['id'] == new['id'] and c['description'] == new['description'] for c in rows)
```

Each complaint in the list should have exactly the fields of `ComplaintOut`.

```py
await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
rows = (await client.get('/api/complaints')).json()
assert len(rows) >= 1
assert all(set(c) == {'id', 'description', 'category', 'urgency', 'status', 'deadline', 'escalation_level', 'created_at', 'resolved_at'} for c in rows)
```

```python file=services/triage.py
# IHMS triage rules, from chapter B4


def deadline_hours(urgency: str) -> int:
    """Return the resolution deadline in hours for an urgency level."""
    if urgency == "urgent":
        return 24
    if urgency == "normal":
        return 72
    return 168


URGENCY_KEYWORDS = {
    "urgent": ["burst", "flood", "spark", "exposed wire", "no water", "broken lock"],
    "normal": ["leaking tap", "faulty socket", "broken bed", "blocked drain"],
    "low": ["paint", "loose handle", "squeaking door"],
}


def triage_urgency(description: str) -> str | None:
    text = description.lower()
    for urgency in ["urgent", "normal", "low"]:      # most serious first
        for keyword in URGENCY_KEYWORDS[urgency]:
            if keyword in text:
                return urgency
    return None   # nothing matched: what happens next is a group decision


CATEGORY_KEYWORDS = {
    "plumbing": ["pipe", "tap", "leak", "water", "drain", "toilet", "shower", "flood"],
    "electrical": ["socket", "light", "bulb", "wire", "spark", "fan", "switch", "power"],
    "carpentry": ["door", "bed", "window", "wardrobe", "handle", "lock", "chair", "table"],
    "sanitation": ["dirty", "smell", "refuse", "waste", "cockroach", "rat", "bin"],
}


def triage_category(description: str) -> str:
    text = description.lower()
    for category in ["plumbing", "electrical", "carpentry", "sanitation"]:
        for keyword in CATEGORY_KEYWORDS[category]:
            if keyword in text:
                return category
    return "other"


# Placeholder until the group decides: the urgency when no keyword matches
NO_KEYWORD_URGENCY = "normal"


def triage(description: str) -> dict:
    category = triage_category(description)
    urgency = triage_urgency(description)
    if urgency is None:
        urgency = NO_KEYWORD_URGENCY
    return {
        "category": category,
        "urgency": urgency,
        "deadline_hours": deadline_hours(urgency),
    }
```
