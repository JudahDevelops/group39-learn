# --description--

Add two more keys. `status` is `"Open"` for a new complaint. `deadline` is `now` plus `timedelta(hours=...)`, with the hours taken from `result["deadline_hours"]`. An urgent complaint has 24 hours, a normal one 72 and a low one 7 days (168 hours). Your `triage` function already knows that.

# --hints--

A new complaint should have the status `Open`.

```py
r = await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
assert r.json()['status'] == 'Open'
```

The deadline of an urgent complaint should be 24 hours from now.

```py
from datetime import datetime, timedelta, timezone
r = await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
left = datetime.fromisoformat(r.json()['deadline']) - datetime.now(timezone.utc)
assert timedelta(hours=23, minutes=58) < left < timedelta(hours=24, minutes=2)
```

The deadline of a normal complaint, such as "The blocked drain is smelling bad", should be 72 hours from now.

```py
from datetime import datetime, timedelta, timezone
r = await client.post('/api/complaints', json={'description': 'The blocked drain is smelling bad'})
assert r.json()['urgency'] == 'normal'
left = datetime.fromisoformat(r.json()['deadline']) - datetime.now(timezone.utc)
assert timedelta(hours=71, minutes=58) < left < timedelta(hours=72, minutes=2)
```

The deadline of a low complaint, such as "Please paint the wall in my room", should be 7 days from now.

```py
from datetime import datetime, timedelta, timezone
r = await client.post('/api/complaints', json={'description': 'Please paint the wall in my room'})
assert r.json()['urgency'] == 'low'
left = datetime.fromisoformat(r.json()['deadline']) - datetime.now(timezone.utc)
assert timedelta(days=6, hours=23, minutes=58) < left < timedelta(days=7, minutes=2)
```

```python file=services/triage.py
# IHMS triage rules, from chapter B4


def deadline_hours(urgency: str) -> int:
    """Return the resolution deadline in hours for an urgency level."""
    if urgency == "urgent":
        return 24
    if urgency == "normal":
        return 72
    return 168


URGENCY_KEYWORDS = {
    "urgent": ["burst", "flood", "spark", "exposed wire", "no water", "broken lock"],
    "normal": ["leaking tap", "faulty socket", "broken bed", "blocked drain"],
    "low": ["paint", "loose handle", "squeaking door"],
}


def triage_urgency(description: str) -> str | None:
    text = description.lower()
    for urgency in ["urgent", "normal", "low"]:      # most serious first
        for keyword in URGENCY_KEYWORDS[urgency]:
            if keyword in text:
                return urgency
    return None   # nothing matched: what happens next is a group decision


CATEGORY_KEYWORDS = {
    "plumbing": ["pipe", "tap", "leak", "water", "drain", "toilet", "shower", "flood"],
    "electrical": ["socket", "light", "bulb", "wire", "spark", "fan", "switch", "power"],
    "carpentry": ["door", "bed", "window", "wardrobe", "handle", "lock", "chair", "table"],
    "sanitation": ["dirty", "smell", "refuse", "waste", "cockroach", "rat", "bin"],
}


def triage_category(description: str) -> str:
    text = description.lower()
    for category in ["plumbing", "electrical", "carpentry", "sanitation"]:
        for keyword in CATEGORY_KEYWORDS[category]:
            if keyword in text:
                return category
    return "other"


# Placeholder until the group decides: the urgency when no keyword matches
NO_KEYWORD_URGENCY = "normal"


def triage(description: str) -> dict:
    category = triage_category(description)
    urgency = triage_urgency(description)
    if urgency is None:
        urgency = NO_KEYWORD_URGENCY
    return {
        "category": category,
        "urgency": urgency,
        "deadline_hours": deadline_hours(urgency),
    }
```
