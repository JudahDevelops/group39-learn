# --description--

Add three more keys. `escalation_level` is `0`: no email has been sent about this complaint yet. `created_at` is `now`. `resolved_at` is `None`, because a new complaint is not resolved. The deadline and `created_at` both come from `now`, so the gap between them is exactly the length of the deadline.

# --hints--

A new complaint should have `escalation_level` 0.

```py
r = await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
assert r.json()['escalation_level'] == 0
```

A new complaint should have `created_at` set to the time it was lodged. Use `now`.

```py
from datetime import datetime, timedelta, timezone
r = await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
created = datetime.fromisoformat(r.json()['created_at'])
assert abs(datetime.now(timezone.utc) - created) < timedelta(minutes=1)
```

A new complaint should have `resolved_at` set to `None`, so the reply has `null` there.

```py
r = await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
assert 'resolved_at' in r.json()
assert r.json()['resolved_at'] is None
```

The deadline should be exactly 24 hours after `created_at` for an urgent complaint. Use the same `now` for both.

```py
from datetime import datetime, timedelta
r = await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})
data = r.json()
assert datetime.fromisoformat(data['deadline']) - datetime.fromisoformat(data['created_at']) == timedelta(hours=24)
```

```python file=services/triage.py
# IHMS triage rules, from chapter B4


def deadline_hours(urgency: str) -> int:
    """Return the resolution deadline in hours for an urgency level."""
    if urgency == "urgent":
        return 24
    if urgency == "normal":
        return 72
    return 168


URGENCY_KEYWORDS = {
    "urgent": ["burst", "flood", "spark", "exposed wire", "no water", "broken lock"],
    "normal": ["leaking tap", "faulty socket", "broken bed", "blocked drain"],
    "low": ["paint", "loose handle", "squeaking door"],
}


def triage_urgency(description: str) -> str | None:
    text = description.lower()
    for urgency in ["urgent", "normal", "low"]:      # most serious first
        for keyword in URGENCY_KEYWORDS[urgency]:
            if keyword in text:
                return urgency
    return None   # nothing matched: what happens next is a group decision


CATEGORY_KEYWORDS = {
    "plumbing": ["pipe", "tap", "leak", "water", "drain", "toilet", "shower", "flood"],
    "electrical": ["socket", "light", "bulb", "wire", "spark", "fan", "switch", "power"],
    "carpentry": ["door", "bed", "window", "wardrobe", "handle", "lock", "chair", "table"],
    "sanitation": ["dirty", "smell", "refuse", "waste", "cockroach", "rat", "bin"],
}


def triage_category(description: str) -> str:
    text = description.lower()
    for category in ["plumbing", "electrical", "carpentry", "sanitation"]:
        for keyword in CATEGORY_KEYWORDS[category]:
            if keyword in text:
                return category
    return "other"


# Placeholder until the group decides: the urgency when no keyword matches
NO_KEYWORD_URGENCY = "normal"


def triage(description: str) -> dict:
    category = triage_category(description)
    urgency = triage_urgency(description)
    if urgency is None:
        urgency = NO_KEYWORD_URGENCY
    return {
        "category": category,
        "urgency": urgency,
        "deadline_hours": deadline_hours(urgency),
    }
```
