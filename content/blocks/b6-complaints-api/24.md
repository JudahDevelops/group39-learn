# --description--

Add a route for `PATCH /{complaint_id}/advance` with `response_model=ComplaintOut`. The function `advance_complaint` finds the complaint with `find_complaint` (a missing one is a 404 with the same detail as before), sets its `status` to `NEXT_STATUS[complaint["status"]]` and returns it. A PATCH changes part of something that exists, so it has no body.

# --hints--

A complaint that does not exist should give status 404 on `PATCH /api/complaints/{id}/advance`.

```py
r = await client.patch('/api/complaints/999999/advance')
assert r.status_code == 404
```

The 404 should have the detail `Complaint not found`.

```py
r = await client.patch('/api/complaints/999999/advance')
assert r.json()['detail'] == 'Complaint not found'
```

The first advance should move an `Open` complaint to `Assigned`, with status 200.

```py
new = (await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})).json()
r = await client.patch('/api/complaints/' + str(new['id']) + '/advance')
assert r.status_code == 200
assert r.json()['status'] == 'Assigned'
```

The next advances should give `In Progress`, then `Resolved`.

```py
new = (await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})).json()
url = '/api/complaints/' + str(new['id']) + '/advance'
assert (await client.patch(url)).json()['status'] == 'Assigned'
assert (await client.patch(url)).json()['status'] == 'In Progress'
assert (await client.patch(url)).json()['status'] == 'Resolved'
```

The reply should have exactly the fields of `ComplaintOut`, and no `history`.

```py
new = (await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})).json()
r = await client.patch('/api/complaints/' + str(new['id']) + '/advance')
assert set(r.json()) == {'id', 'description', 'category', 'urgency', 'status', 'deadline', 'escalation_level', 'created_at', 'resolved_at'}
```

The change should be kept: the complaint should show under `?status=Assigned` and no longer under `?status=Open`.

```py
new = (await client.post('/api/complaints', json={'description': 'The pipe in the bathroom has burst'})).json()
await client.patch('/api/complaints/' + str(new['id']) + '/advance')
assigned = [c['id'] for c in (await client.get('/api/complaints', params={'status': 'Assigned'})).json()]
open_ids = [c['id'] for c in (await client.get('/api/complaints', params={'status': 'Open'})).json()]
assert new['id'] in assigned
assert new['id'] not in open_ids
```

```python file=routers/complaints.py
# The complaint endpoints.

from datetime import datetime, timedelta, timezone

from fastapi import APIRouter, HTTPException

from schemas import ComplaintDetail, ComplaintIn, ComplaintOut
from services.triage import triage

router = APIRouter(prefix="/complaints", tags=["complaints"])

COMPLAINTS = []


def by_deadline(complaint):
    return complaint["deadline"]


@router.get("", response_model=list[ComplaintOut])
def list_complaints(status: str | None = None):
    complaints = COMPLAINTS
    if status is not None:
        complaints = [c for c in complaints if c["status"] == status]
    return sorted(complaints, key=by_deadline)


@router.post("", response_model=ComplaintOut, status_code=201)
def create_complaint(body: ComplaintIn):
    result = triage(body.description)
    now = datetime.now(timezone.utc)
    complaint = {
        "id": len(COMPLAINTS) + 1,
        "description": body.description,
        "category": result["category"],
        "urgency": result["urgency"],
        "status": "Open",
        "deadline": now + timedelta(hours=result["deadline_hours"]),
        "escalation_level": 0,
        "created_at": now,
        "resolved_at": None,
        "history": [
            {
                "event": f"Lodged and triaged as {result['urgency']} {result['category']}",
                "actor_id": None,
                "at": now,
            }
        ],
    }
    COMPLAINTS.append(complaint)
    return complaint


def find_complaint(complaint_id: int):
    for complaint in COMPLAINTS:
        if complaint["id"] == complaint_id:
            return complaint
    return None


@router.get("/{complaint_id}", response_model=ComplaintDetail)
def get_complaint(complaint_id: int):
    complaint = find_complaint(complaint_id)
    if complaint is None:
        raise HTTPException(status_code=404, detail="Complaint not found")
    return complaint


NEXT_STATUS = {"Open": "Assigned", "Assigned": "In Progress", "In Progress": "Resolved"}

# IHMS triage rules, from chapter B4


def deadline_hours(urgency: str) -> int:
    """Return the resolution deadline in hours for an urgency level."""
    if urgency == "urgent":
        return 24
    if urgency == "normal":
        return 72
    return 168


URGENCY_KEYWORDS = {
    "urgent": ["burst", "flood", "spark", "exposed wire", "no water", "broken lock"],
    "normal": ["leaking tap", "faulty socket", "broken bed", "blocked drain"],
    "low": ["paint", "loose handle", "squeaking door"],
}


def triage_urgency(description: str) -> str | None:
    text = description.lower()
    for urgency in ["urgent", "normal", "low"]:      # most serious first
        for keyword in URGENCY_KEYWORDS[urgency]:
            if keyword in text:
                return urgency
    return None   # nothing matched: what happens next is a group decision


CATEGORY_KEYWORDS = {
    "plumbing": ["pipe", "tap", "leak", "water", "drain", "toilet", "shower", "flood"],
    "electrical": ["socket", "light", "bulb", "wire", "spark", "fan", "switch", "power"],
    "carpentry": ["door", "bed", "window", "wardrobe", "handle", "lock", "chair", "table"],
    "sanitation": ["dirty", "smell", "refuse", "waste", "cockroach", "rat", "bin"],
}


def triage_category(description: str) -> str:
    text = description.lower()
    for category in ["plumbing", "electrical", "carpentry", "sanitation"]:
        for keyword in CATEGORY_KEYWORDS[category]:
            if keyword in text:
                return category
    return "other"


# Placeholder until the group decides: the urgency when no keyword matches
NO_KEYWORD_URGENCY = "normal"


def triage(description: str) -> dict:
    category = triage_category(description)
    urgency = triage_urgency(description)
    if urgency is None:
        urgency = NO_KEYWORD_URGENCY
    return {
        "category": category,
        "urgency": urgency,
        "deadline_hours": deadline_hours(urgency),
    }
```
