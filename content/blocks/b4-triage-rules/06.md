# --description--

From the deadline until twice the deadline, the level is 1. Below the first `if`, add a second one that returns `1` when `elapsed_hours` is less than `2 * deadline_hours`. The first `if` has already returned for anything under the deadline, so this one only sees times that have reached it.

def deadline_hours(urgency: str) -> int:
    """Return the resolution deadline in hours for an urgency level."""
    if urgency == "urgent":
        return 24
    if urgency == "normal":
        return 72
    return 168


def escalation_level(elapsed_hours: float, deadline_hours: float) -> int:
    if elapsed_hours < deadline_hours:
        return 0
    if elapsed_hours < 2 * deadline_hours:
        return 1
