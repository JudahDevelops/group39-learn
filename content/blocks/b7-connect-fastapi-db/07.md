# --description--

Add the three time columns. `created_at` and `deadline` are moments in time with their time zone: `mapped_column(DateTime(timezone=True))`. `resolved_at` is the same, but empty until the complaint is resolved, so its type is `Mapped[datetime | None]`. `datetime` and `DateTime` are already imported at the top.

# --hints--

`created_at` and `deadline` should be `DateTime(timezone=True)` columns that must have a value.

```py
import models
from sqlalchemy import DateTime

for name in ('created_at', 'deadline'):
    assert name in cols
    assert isinstance(cols[name].type, DateTime)
    assert cols[name].type.timezone is True
    assert not cols[name].nullable
```

```py

from sqlalchemy import DateTime

assert 'resolved_at' in cols
```

A saved `Complaint` should start with the status `Open`, the escalation level `0` and no `resolved_at`.

```py
from datetime import datetime, timedelta, timezone
import database, models
database.Base.metadata.create_all(database.engine)
now = datetime.now(timezone.utc)
with database.SessionLocal() as db:
    c = models.Complaint(student_id=1, room_id=1, description='The pipe in the bathroom has burst', category='plumbing', urgency='urgent', created_at=now, deadline=now + timedelta(hours=24))
    db.add(c)
    db.commit()
    cid = c.id
with database.SessionLocal() as db:
    c = db.get(models.Complaint, cid)
    assert c.status == 'Open'
    assert c.escalation_level == 0
    assert c.resolved_at is None
```

```python file=services/triage.py
# IHMS triage rules, from chapter B4


def deadline_hours(urgency: str) -> int:
    """Return the resolution deadline in hours for an urgency level."""
    if urgency == "urgent":
        return 24
    if urgency == "normal":
        return 72
    return 168


URGENCY_KEYWORDS = {
    "urgent": ["burst", "flood", "spark", "exposed wire", "no water", "broken lock"],
    "normal": ["leaking tap", "faulty socket", "broken bed", "blocked drain"],
    "low": ["paint", "loose handle", "squeaking door"],
}


def triage_urgency(description: str) -> str | None:
    text = description.lower()
    for urgency in ["urgent", "normal", "low"]:      # most serious first
        for keyword in URGENCY_KEYWORDS[urgency]:
            if keyword in text:
                return urgency
    return None   # nothing matched: what happens next is a group decision


CATEGORY_KEYWORDS = {
    "plumbing": ["pipe", "tap", "leak", "water", "drain", "toilet", "shower", "flood"],
    "electrical": ["socket", "light", "bulb", "wire", "spark", "fan", "switch", "power"],
    "carpentry": ["door", "bed", "window", "wardrobe", "handle", "lock", "chair", "table"],
    "sanitation": ["dirty", "smell", "refuse", "waste", "cockroach", "rat", "bin"],
}


def triage_category(description: str) -> str:
    text = description.lower()
    for category in ["plumbing", "electrical", "carpentry", "sanitation"]:
        for keyword in CATEGORY_KEYWORDS[category]:
            if keyword in text:
                return category
    return "other"
```

```python file=services/escalation.py
    return created_at + timedelta(hours=deadline_hours(urgency))
```
