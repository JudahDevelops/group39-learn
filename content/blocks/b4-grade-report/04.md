```python file=main.py
# Grade report for COS 311
print("Grade Report: COS 311")

students = [
    {"name": "Amina Bello", "score": 78},
    {"name": "Chidi Okafor", "score": 64},
    {"name": "Bola Adeyemi", "score": 55},
    {"name": "Emeka Eze", "score": 47},
    {"name": "Ngozi Obi", "score": 42},
    {"name": "Fatima Yusuf", "score": 35},
]

for student in students:
    print(student)
```
