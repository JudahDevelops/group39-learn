```python file=main.py
# Grade report for COS 311
print("Grade Report: COS 311")

students = [
    {"name": "Amina Bello", "score": 78},
    {"name": "Chidi Okafor", "score": 64},
    {"name": "Bola Adeyemi", "score": 55},
    {"name": "Emeka Eze", "score": 47},
    {"name": "Ngozi Obi", "score": 42},
    {"name": "Fatima Yusuf", "score": 35},
]


def grade_for(score: float) -> str:
    """Return the letter grade for a score from 0 to 100."""
    if score >= 70:
        return "A"
    if score >= 60:
        return "B"
    if score >= 50:
        return "C"
    if score >= 45:
        return "D"
    if score >= 40:
        return "E"
    return "F"


total = 0
for student in students:
    name = student["name"]
    score = student["score"]
    grade = grade_for(score)
    print(f"{name}: {score} ({grade})")
    total += score
print(f"Total score: {total}")

average = total / len(students)
```
