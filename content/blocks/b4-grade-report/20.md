]


def grade_for(score: float) -> str:
    """Return the letter grade for a score from 0 to 100."""
    if score >= 70:
        return "A"
    if score >= 60:
        return "B"
    if score >= 50:
        return "C"
    if score >= 45:
        return "D"
    if score >= 40:
        return "E"
    return "F"


def report_line(name: str, score: float) -> str:
    """Return one line of the report."""
    grade = grade_for(score)
    return f"{name}: {score} ({grade})"


total = 0
