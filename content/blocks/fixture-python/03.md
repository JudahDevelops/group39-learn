```python file=main.py
# Notice for the hall board
print("Water is off in Hall A today.")

def compute_deadline(urgency):
    if urgency == "urgent":
        return 24
    elif urgency == "normal":
        return 72
    else:
        return 168

print(f"Deadline: {compute_deadline('urgent')} hours")
```
