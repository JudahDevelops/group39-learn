# --description--

When `resetButton` is clicked, empty both inputs and both paragraphs. Setting an input's `.value` to `""` empties it, and so does setting a paragraph's `textContent` to `""`.

# --hints--

Clicking the reset button should empty the bill field and the people field.

```js
document.querySelector('#bill').value = '25000';
document.querySelector('#people').value = '3';
document.querySelector('#split').click();
document.querySelector('#reset').click();
assert.equal(document.querySelector('#bill').value, '');
assert.equal(document.querySelector('#people').value, '');
```

Clicking the reset button should empty the `result` and `together` paragraphs.

```js
document.querySelector('#bill').value = '25000';
document.querySelector('#people').value = '3';
document.querySelector('#split').click();
document.querySelector('#reset').click();
assert.equal(document.querySelector('#result').textContent.trim(), '');
assert.equal(document.querySelector('#together').textContent.trim(), '');
```

Your split button should still work after a reset. `10000` shared by `4` should show `Each person pays ₦2500`.

```js
document.querySelector('#bill').value = '25000';
document.querySelector('#people').value = '3';
document.querySelector('#split').click();
document.querySelector('#reset').click();
document.querySelector('#bill').value = '10000';
document.querySelector('#people').value = '4';
document.querySelector('#split').click();
assert.equal(document.querySelector('#result').textContent.trim(), 'Each person pays ₦2500');
```

console.log("Bill Splitter ready");

function splitBill(total, count) {
  return Math.round(total / count);
}

const billInput = document.querySelector("#bill");
const peopleInput = document.querySelector("#people");
const splitButton = document.querySelector("#split");
const result = document.querySelector("#result");
const together = document.querySelector("#together");
const resetButton = document.querySelector("#reset");

function showShare() {
  const total = Number(billInput.value);
  const count = Number(peopleInput.value);

  if (total <= 0) {
    result.textContent = "Enter a bill total above zero.";
    together.textContent = "";
    return;
  }

  if (count <= 0) {
    result.textContent = "Enter a number of people above zero.";
    together.textContent = "";
    return;
  }

  const share = splitBill(total, count);
  result.textContent = `Each person pays ₦${share}`;
  together.textContent = `Together the shares add up to ₦${share * count}`;
}

splitButton.addEventListener("click", showShare);
peopleInput.addEventListener("change", showShare);

console.log("Bill Splitter ready");

function splitBill(total, count) {
  return Math.round(total / count);
}

const billInput = document.querySelector("#bill");
const peopleInput = document.querySelector("#people");
const splitButton = document.querySelector("#split");
const result = document.querySelector("#result");
const together = document.querySelector("#together");
const resetButton = document.querySelector("#reset");

function showShare() {
  const total = Number(billInput.value);
  const count = Number(peopleInput.value);

  if (total <= 0) {
    result.textContent = "Enter a bill total above zero.";
    together.textContent = "";
    return;
  }

  if (count <= 0) {
    result.textContent = "Enter a number of people above zero.";
    together.textContent = "";
    return;
  }

  const share = splitBill(total, count);
  result.textContent = `Each person pays ₦${share}`;
  together.textContent = `Together the shares add up to ₦${share * count}`;

splitButton.addEventListener("click", showShare);
peopleInput.addEventListener("change", showShare);

resetButton.addEventListener("click", () => {
  billInput.value = "";
  peopleInput.value = "";
  result.textContent = "";
  together.textContent = "";
});
