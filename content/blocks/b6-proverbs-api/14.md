# --description--

Add a route for `POST /api/proverbs` with `status_code=201`. The function takes `body: ProverbIn`. Build a new proverb dictionary with `id` from `next_id()` and the `text`, `language` and `meaning` from the body. Append it to `PROVERBS` and return it.

# --hints--

`POST /api/proverbs` should answer with status 201. Add `status_code=201` to the decorator.

```py
r = await client.post('/api/proverbs', json={"text": "Let the kite perch and let the eagle perch.", "language": "Igbo", "meaning": "Live and let live: there is room for everyone."})
assert r.status_code == 201
```

The reply should hold the proverb you sent: its `text`, `language` and `meaning`.

```py
sent = {"text": "Let the kite perch and let the eagle perch.", "language": "Igbo", "meaning": "Live and let live: there is room for everyone."}
r = await client.post('/api/proverbs', json=sent)
for key in sent:
    assert r.json()[key] == sent[key]
```

The reply should have an `id` that no earlier proverb had. Use `next_id()`.

```py
before = {p['id'] for p in (await client.get('/api/proverbs')).json()}
r = await client.post('/api/proverbs', json={"text": "Let the kite perch and let the eagle perch.", "language": "Igbo", "meaning": "Live and let live: there is room for everyone."})
assert isinstance(r.json()['id'], int)
assert r.json()['id'] not in before
```

The new proverb should be added to `PROVERBS`, so that it appears in `GET /api/proverbs`.

```py
before = len((await client.get('/api/proverbs')).json())
await client.post('/api/proverbs', json={"text": "Let the kite perch and let the eagle perch.", "language": "Igbo", "meaning": "Live and let live: there is room for everyone."})
after = len((await client.get('/api/proverbs')).json())
assert after == before + 1
```

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field

from proverbs import PROVERBS

app = FastAPI(title="Proverbs API")


class ProverbIn(BaseModel):
    text: str
    language: str
    meaning: str


def next_id() -> int:
    highest = 0
    for proverb in PROVERBS:
        if proverb["id"] > highest:
            highest = proverb["id"]
    return highest + 1


def find_proverb(proverb_id: int):
    for proverb in PROVERBS:
        if proverb["id"] == proverb_id:
            return proverb
    return None


@app.get("/api/health")
def health():
    return {"status": "ok"}


@app.get("/api/proverbs")
def list_proverbs(language: str | None = None):
    if language is None:
        return PROVERBS
    return [p for p in PROVERBS if p["language"].lower() == language.lower()]


@app.get("/api/proverbs/{proverb_id}")
def get_proverb(proverb_id: int):
    proverb = find_proverb(proverb_id)
    if proverb is None:
        raise HTTPException(status_code=404, detail="Proverb not found")
    return proverb

```python file=main.py
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field

from proverbs import PROVERBS

app = FastAPI(title="Proverbs API")


class ProverbIn(BaseModel):
    text: str
    language: str
    meaning: str


def next_id() -> int:
    highest = 0
    for proverb in PROVERBS:
        if proverb["id"] > highest:
            highest = proverb["id"]
    return highest + 1


def find_proverb(proverb_id: int):
    for proverb in PROVERBS:
        if proverb["id"] == proverb_id:
            return proverb
    return None


@app.get("/api/health")
def health():
    return {"status": "ok"}


@app.get("/api/proverbs")
def list_proverbs(language: str | None = None):
    if language is None:
        return PROVERBS
    return [p for p in PROVERBS if p["language"].lower() == language.lower()]


@app.get("/api/proverbs/{proverb_id}")
def get_proverb(proverb_id: int):
    proverb = find_proverb(proverb_id)
    if proverb is None:
        raise HTTPException(status_code=404, detail="Proverb not found")
    return proverb


@app.post("/api/proverbs", status_code=201)
def create_proverb(body: ProverbIn):
    proverb = {
        "id": next_id(),
        "text": body.text,
        "language": body.language,
        "meaning": body.meaning,
    }
    PROVERBS.append(proverb)
    return proverb
```

## --hint--

The argument `body: ProverbIn` tells FastAPI to read the request body as a `ProverbIn`. Inside the function you read its fields with a dot, as in `body.text`.

## --hint--

Build a dictionary with the four keys `id`, `text`, `language` and `meaning`. Call `next_id()` for the `id`. Then `PROVERBS.append(proverb)` and return it.
